```bash
require_fixed() {
  local file=$1
  local needle=$2
  grep -Fq -- "$needle" "$file" || fail "$file is missing: $needle"
}

require_absent_regex() {
  local file=$1
  local pattern=$2
  if grep -Eq -- "$pattern" "$file"; then
    fail "$file must not match: $pattern"
  fi
}
```

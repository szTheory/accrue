```bash
fail() {
  echo "[verify_package_docs] package docs verification failed: $*" >&2
  exit 1
}

require_fixed() { grep -Fq "$2" "$1" || fail "$1 is missing: $2"; }
require_regex() { grep -Eq "$2" "$1" || fail "$1 does not match: $2"; }
require_absent_regex() {
  if grep -Eq "$2" "$1"; then
    fail "$1 must not match: $2"
  fi
}
```
